import cv2
import numpy as np

# Read the image
img = cv2.imread("flower.jpeg")

# Check if image is loaded
if img is None:
    print("Error: Image not found!")
    exit()

# Get image size
rows, cols = img.shape[:2]

# Move the image
tx = 100   # Move 100 pixels right
ty = 50    # Move 50 pixels down

# Translation matrix
M = np.float32([[1, 0, tx],
                [0, 1, ty]])

# Apply translation
moved = cv2.warpAffine(img, M, (cols, rows))

# Save output image
cv2.imwrite("moved_image.jpg", moved)

print("Image moved successfully!")
