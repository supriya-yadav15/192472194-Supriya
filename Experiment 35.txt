import cv2
import numpy as np
import os

# Create output folder automatically
output_folder = "outputs"
os.makedirs(output_folder, exist_ok=True)

# Read input image
image = cv2.imread("flower.jpeg", cv2.IMREAD_GRAYSCALE)

if image is None:
    print("Error: flower.jpeg not found!")
    exit()

# Define morphological kernel
kernel = np.ones((5, 5), np.uint8)

# Apply Black Hat
# Black Hat = Closing - Original Image
black_hat = cv2.morphologyEx(
    image,
    cv2.MORPH_BLACKHAT,
    kernel
)

# Save output image
output_path = "outputs/exp35_black_hat.jpg"

success = cv2.imwrite(output_path, black_hat)

# Display result information
print("----------------------------------------")
print("Experiment 35: Morphological Black Hat")
print("----------------------------------------")
print("Input Image : flower.jpeg")
print("Output Image:", output_path)

if success:
    print("Experiment 35 completed successfully.")
else:
    print("Error: Output image was not saved.")

print("----------------------------------------")
