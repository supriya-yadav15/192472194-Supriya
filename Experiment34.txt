import cv2
import numpy as np
import os

# Create output folder automatically
output_folder = "outputs"
os.makedirs(output_folder, exist_ok=True)

# Read input image
image = cv2.imread("flower.jpeg", cv2.IMREAD_GRAYSCALE)

if image is None:
    print("Error: flower.jpeg not found!")
    exit()

# Define morphological kernel
kernel = np.ones((5, 5), np.uint8)

# Apply Top Hat
# Top Hat = Original Image - Opening
top_hat = cv2.morphologyEx(
    image,
    cv2.MORPH_TOPHAT,
    kernel
)

# Save output image
output_path = "outputs/exp34_top_hat.jpg"

success = cv2.imwrite(output_path, top_hat)

# Display result information
print("----------------------------------------")
print("Experiment 34: Morphological Top Hat")
print("----------------------------------------")
print("Input Image : flower.jpeg")
print("Output Image:", output_path)

if success:
    print("Experiment 34 completed successfully.")
else:
    print("Error: Output image was not saved.")

print("----------------------------------------")
