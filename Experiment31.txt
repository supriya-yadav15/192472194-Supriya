import cv2
import numpy as np
import os

# -----------------------------------------
# Create output folder automatically
# -----------------------------------------
output_folder = "outputs"
os.makedirs(output_folder, exist_ok=True)

# -----------------------------------------
# Read input image
# -----------------------------------------
image = cv2.imread("flower.jpeg", cv2.IMREAD_GRAYSCALE)

if image is None:
    print("Error: flower.jpeg not found!")
    exit()

# -----------------------------------------
# Define morphological kernel
# -----------------------------------------
kernel = np.ones((5, 5), np.uint8)

# -----------------------------------------
# Apply Opening
# Opening = Erosion followed by Dilation
# -----------------------------------------
opening = cv2.morphologyEx(
    image,
    cv2.MORPH_OPEN,
    kernel
)

# -----------------------------------------
# Save output image
# -----------------------------------------
output_path = "outputs/exp31_opening.jpg"

cv2.imwrite(output_path, opening)

# -----------------------------------------
# Display result information
# -----------------------------------------
print("----------------------------------------")
print("Experiment 31: Morphological Opening")
print("----------------------------------------")
print("Input Image : flower.jpeg")
print("Output Image:", output_path)
print("Experiment 31 completed successfully.")
print("----------------------------------------")
