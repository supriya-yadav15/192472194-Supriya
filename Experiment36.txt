import cv2
import os

# -----------------------------------------
# Create output folder automatically
# -----------------------------------------
output_folder = "outputs"
os.makedirs(output_folder, exist_ok=True)

# -----------------------------------------
# Input watch image
# -----------------------------------------
input_image = "watch image.jpeg"

image = cv2.imread(input_image)

if image is None:
    print("Error: watch image.jpeg not found!")
    exit()

# -----------------------------------------
# Convert image to grayscale
# -----------------------------------------
gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)

# -----------------------------------------
# Reduce noise
# -----------------------------------------
blur = cv2.GaussianBlur(gray, (5, 5), 0)

# -----------------------------------------
# Detect edges
# -----------------------------------------
edges = cv2.Canny(blur, 50, 150)

# -----------------------------------------
# Find contours
# -----------------------------------------
contours, _ = cv2.findContours(
    edges,
    cv2.RETR_EXTERNAL,
    cv2.CHAIN_APPROX_SIMPLE
)

# -----------------------------------------
# Find the largest suitable object
# -----------------------------------------
if len(contours) == 0:
    print("No object detected.")
    exit()

largest_contour = max(contours, key=cv2.contourArea)

x, y, w, h = cv2.boundingRect(largest_contour)

# -----------------------------------------
# Draw rectangle around the watch
# -----------------------------------------
cv2.rectangle(
    image,
    (x, y),
    (x + w, y + h),
    (0, 255, 0),
    3
)

# -----------------------------------------
# Add Watch label
# -----------------------------------------
cv2.putText(
    image,
    "WATCH",
    (x, max(y - 10, 30)),
    cv2.FONT_HERSHEY_SIMPLEX,
    1,
    (0, 255, 0),
    2
)

# -----------------------------------------
# Save output
# -----------------------------------------
output_path = "outputs/exp36_watch_recognition.jpg"

success = cv2.imwrite(output_path, image)

# -----------------------------------------
# Display information
# -----------------------------------------
print("----------------------------------------")
print("Experiment 36: Watch Recognition")
print("----------------------------------------")
print("Input Image :", input_image)
print("Output Image:", output_path)

if success:
    print("Watch recognized successfully.")
    print("Output saved successfully.")
else:
    print("Error: Output image could not be saved.")

print("----------------------------------------")
