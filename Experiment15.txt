import cv2
import numpy as np

# Read the input image
img = cv2.imread("flower.jpeg")

# Check whether image is loaded
if img is None:
    print("Error: flower.jpeg not found!")
    exit()

# Resize image
img = cv2.resize(img, (500, 500))

# Source points
src_points = np.float32([
    [50, 50],
    [450, 50],
    [450, 450],
    [50, 450]
])

# Destination points
dst_points = np.float32([
    [100, 100],
    [400, 50],
    [450, 450],
    [50, 400]
])

# Compute Perspective Transformation Matrix (DLT)
matrix = cv2.getPerspectiveTransform(src_points, dst_points)

# Apply the transformation
output = cv2.warpPerspective(img, matrix, (500, 500))

# Save output image
cv2.imwrite("dlt_output.jpg", output)

print("Direct Linear Transformation completed successfully.")
print("Output saved as dlt_output.jpg")
