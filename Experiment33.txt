import cv2
import numpy as np
import os

# Create output folder automatically
output_folder = "outputs"
os.makedirs(output_folder, exist_ok=True)

# Read input image
image = cv2.imread("flower.jpeg", cv2.IMREAD_GRAYSCALE)

if image is None:
    print("Error: flower.jpeg not found!")
    exit()

# Define morphological kernel
kernel = np.ones((5, 5), np.uint8)

# Apply Morphological Gradient
# Gradient = Dilation - Erosion
gradient = cv2.morphologyEx(
    image,
    cv2.MORPH_GRADIENT,
    kernel
)

# Save output image
output_path = "outputs/exp33_morphological_gradient.jpg"

success = cv2.imwrite(output_path, gradient)

# Display result information
print("----------------------------------------")
print("Experiment 33: Morphological Gradient")
print("----------------------------------------")
print("Input Image : flower.jpeg")
print("Output Image:", output_path)

if success:
    print("Experiment 33 completed successfully.")
else:
    print("Error: Output image was not saved.")

print("----------------------------------------")
